  async function request(path, opts = {}) {
    const headers = {"accept": "application/json", ...(opts.body ? {"content-type": "application/json"} : {}), ...opts.headers};
    const res = await fetch(path, {...opts, headers, cache: "no-store", credentials: "omit"});
    let payload = null;
    try { payload = await res.json(); } catch (_) {}
    if (!res.ok) { const failure = new Error(payload?.error || "Request failed (HTTP " + res.status + ")"); failure.httpStatus = res.status; throw failure; }
    return payload;
  }

    const noticeItem=rosterNoticeQueue.shift();

      noticeItem.kind==="moved"?"SWITCHED TO":"REMOVED FROM";
    toast.replaceChildren();

      noticeItem.nickname+" "+label+" "+noticeItem.team));
    toast.hidden=false;
    // Every distinct join/leave is shown, even if several updates arrive in
    // one WebSocket snapshot. A slow mobile connection cannot erase a name.
    presenceDismiss=setTimeout(()=>{
      presenceDismiss=null;showNextRosterNotice();
    },4000);

  function renderTeamClocks() {
    if (!state || state.phase !== "advisers" || !state.clock) return;
    const now = Date.now() + clockOffset, clock = state.clock;
    const format = ms => {
      const sec = Math.ceil(Math.max(0,ms) / 1000);
      return String(Math.floor(sec/60)).padStart(2,"0") + ":" +
        String(sec%60).padStart(2,"0");
    };
    const nodes = document.querySelectorAll("[data-team-clock]");
    nodes.forEach(el => {
      const i = Number(el.dataset.teamClock), team = state.teams[i];
      if (!team) return;
      if (team.timedOut) {
        el.textContent = "TIME EXPIRED";return;
      }
      const lockedAt = team.lockedAtMs;
      if (team.locked) {
        const time = Number.isSafeInteger(lockedAt) ?
          Math.max(0,(lockedAt < clock.normalEndsAt ? clock.normalEndsAt : clock.closesAt) - lockedAt) : 0;
        el.textContent = "CLOCK STOPPED · " + format(time) + " LEFT";
      } else {
        const grace = now >= clock.normalEndsAt;
        const remaining = Math.max(0,(grace ? clock.closesAt : clock.normalEndsAt) - now);
        el.textContent = (grace ? "RESERVE · " : "DECIDING · ") + format(remaining);
      }
    });
  }

  function renderTeamAftershock(impact) {
    // Read the full consequence when a decision locks; collapse the old file
    // next episode so a three-minute ballot stays ahead of retrospective reading.
    const report = node("details", null, "aftershock-report");
    report.setAttribute("aria-label", "Private consequence of your last team decision");
    report.open = impact.round === state.round + 1;

      "aftershock-title"));
    report.append(summary);
    report.append(node("p", impact.fallout, "aftershock-fallout"));
    const ledger = node("div", null, "aftershock-ledger");
    const trust = impact.trustChange;
    for (const [label,value] of [
      ["TOTAL PAID THIS EPISODE","£" + impact.roundSpent.toLocaleString("en-GB")],
      ["PARTNER TRUST CHANGE",(trust>0?"+":"") + trust + " pts"],
      ["REMAINING COMMITMENTS","£" + impact.pendingLiability.toLocaleString("en-GB")]
    ]) {
      const entry=node("div",null,"aftershock-stat");
      entry.append(node("small",label),node("strong",value));ledger.append(entry);
    }
    report.append(ledger);
    if(impact.paidDeferred.length){
      report.append(node("p","PREVIOUS PROMISES PAID THIS EPISODE: "+
        impact.paidDeferred.join("; "),"aftershock-deferred"));
    }
    if(Number.isFinite(impact.cash)){
      report.append(node("p",impact.cash<0?
        "FUNDING POSITION AT EPISODE CLOSE: £"+Math.abs(impact.cash).toLocaleString("en-GB")+
          " below zero; further promises are not already funded.":
        "RESPONSE CASH AT EPISODE CLOSE: £"+impact.cash.toLocaleString("en-GB")+
          ". Future commitments shown above are still obligations, not available cash.",
        "aftershock-context-line"));
    }
    if(impact.risks&&impact.riskChange){
      const exposure=[["legal","Legal"],["ops","Operations"],["workforce","Workforce"],["customer","Customer"]];
      const rising=exposure.filter(([key])=>impact.riskChange[key]>0).sort((a,b)=>
        impact.riskChange[b[0]]-impact.riskChange[a[0]])[0];
      const falling=exposure.filter(([key])=>impact.riskChange[key]<0).sort((a,b)=>
        impact.riskChange[a[0]]-impact.riskChange[b[0]])[0];
      const statements=[];
      if(falling)statements.push(falling[1]+" exposure fell "+Math.abs(impact.riskChange[falling[0]])+
        " points to "+impact.risks[falling[0]]+".");
      if(rising)statements.push(rising[1]+" exposure rose "+impact.riskChange[rising[0]]+
        " points to "+impact.risks[rising[0]]+".");
      if(impact.risks.workforce>=68)statements.push("Workforce exposure remains high; capacity promises need care.");
      if(impact.trust<35)statements.push("Partner confidence is below 35/100; further assurances need evidence.");
      if(statements.length)report.append(node("p",
        "MEASURED AFTER THIS EPISODE: "+statements.join(" ")+" Changes include automatic episode pressures and previous commitments.",
        "aftershock-context-line"));
    }
    return report;
  }

  function renderCashTrajectory(team) {
    const wrap=node("section",null,"cash-trajectory");
    wrap.append(node("strong","RESPONSE FUND / CASH TRAJECTORY","cash-heading"));
    const cashValues=[260000,...team.decisions.map(x=>x.cash)];
    const formatCash=n=>"£"+n.toLocaleString("en-GB");
    const svg=document.createElementNS("http://www.w3.org/2000/svg","svg");
    svg.setAttribute("viewBox","0 0 600 112");
    svg.setAttribute("role","img");
    svg.setAttribute("aria-label","Cash by episode, starting with "+formatCash(cashValues[0])+
      ": "+cashValues.slice(1).map((x,i)=>"Episode "+(i+1)+" "+formatCash(x)).join("; "));
    svg.setAttribute("style","display:block;width:100%;height:auto;max-height:120px");
    const draw=(tag,attrs)=>{
      const item=document.createElementNS("http://www.w3.org/2000/svg",tag);
      for(const [k,v]of Object.entries(attrs))item.setAttribute(k,String(v));
      svg.append(item);
    };
    const min=Math.min(0,...cashValues),max=Math.max(260000,...cashValues);
    const range=Math.max(1,max-min);
    const points=cashValues.map((value,i)=>({x:18+i*(564/(cashValues.length-1)),
      y:14+(max-value)/range*82,value}));
    for(let i=0;i<4;i++)draw("line",{x1:12,y1:14+i*27,x2:588,y2:14+i*27,
      stroke:"#b1c6ca",opacity:.28,"stroke-width":1});
    const pointsStr=points.map(p=>p.x.toFixed(1)+","+p.y.toFixed(1)).join(" ");
    draw("polyline",{points:pointsStr,fill:"none",stroke:"#b77b48",
      "stroke-width":3.8,"stroke-linecap":"round","stroke-linejoin":"round"});
    for(const p of points)draw("circle",{cx:p.x,cy:p.y,r:4,fill:"#b77b48",stroke:"#fff","stroke-width":1.5});
    wrap.append(svg);
    const legend=node("div",null,"cash-trajectory-labels");
    legend.append(node("span","OPENING "+formatCash(cashValues[0])));
    legend.append(node("span","CLOSING "+formatCash(cashValues[cashValues.length-1])));
    wrap.append(legend);
    return wrap;
  }

  function tick() {
    if (!state || state.phase !== "advisers" || !state.clock) {
      $("timer-wrap").hidden = true;return;
    }
    $("timer-wrap").hidden = false;
    const teamClock = state.viewer.role === "player" ? state.privateTeam?.teamClock : null;
    const clock = teamClock || state.clock;
    const now = Date.now() + clockOffset;
    const stopped = clock.stage === "locked";
    const finished = clock.stage === "closed" && state.viewer.role === "player";
    const grace = !stopped && now >= clock.normalEndsAt;
    const remain = stopped ? clock.remainingMs :
      Math.max(0, (grace ? clock.closesAt : clock.normalEndsAt) - now);
    const seconds = Math.ceil(remain / 1000);
    $("timer").textContent = String(Math.floor(seconds / 60)).padStart(2, "0") +
      ":" + String(seconds % 60).padStart(2, "0");
    $("decision-dock-clock").textContent=$("timer").textContent;
    $("timer-phase").textContent = stopped ? "TEAM CLOCK STOPPED / RESPONSE LOCKED" :
      finished || now >= clock.closesAt ? "TIME EXPIRED" :
      grace ? "RESERVE MINUTE / −5 PTS" : "STANDARD TIME / 3 MIN";
    $("timer-note").textContent = stopped ?
      "Your team is finished. No more time penalty while the other teams deliberate." :
      now >= clock.closesAt ?
      "Unanswered teams receive 12 penalty points and economic consequences." :
      grace ? "A winning team response confirmed now incurs 5 penalty points." :
      "All teammates must vote and reach a winning response before your timer stops.";
    $("timer-wrap").classList.toggle("grace", grace);
    $("timer-wrap").classList.toggle("stopped", stopped);
    renderTeamClocks();
    if(state.viewer.role==="projector" && state.interventions?.length){
      const latest=state.interventions.at(-1);
      $("projector-intervention").hidden=now-latest.atMs>90000;
    }
    if(state.viewer.role==="host"){
      for(const el of document.querySelectorAll("[data-remove-after]"))
        el.disabled=now<Number(el.dataset.removeAfter);
    }
  }
  async function action(command) {
    error("");
    // A complete individual vote belongs to a single episode and either the
    // original ballot or a particular private runoff. Never silently migrate
    // a delayed vote to a later stage while other players act concurrently.
    const voteScope=command.type==="lock_vote"?{
      expectedRound:state.round,
      expectedRunoffOrdinal:state.privateTeam?.voting?.runoff?.ordinal||0
    }:{};
    // Host/roster changes remain strictly revision-guarded. An independent
    // correctly-scoped vote can bypass unrelated global revision changes.
    for (let attempt = 0; attempt < 2; attempt++) {
      try {
        const response = await request("/api/rooms/" + roomCode + "/actions", {
          method: "POST", headers: {authorization: "Bearer " + credential},
          body: JSON.stringify({...command,...voteScope, expectedRevision: state.revision})
        });
        if (command.type === "select_advisers"){
          draftPicks = null;adviserEditOpen=false;
        }
        showRoom(response.state);
        return;
      } catch (e) {
        const retryable = e.httpStatus === 409 &&
          e.message.includes("Room changed; refresh state and retry") &&
          ["select_advisers", "rename_team", "change_team", "lock_advisers", "submit_decision", "lock_vote", "start", "advance", "set_captain", "remove_player"].includes(command.type);
        try { showRoom((await getState()).state); } catch (_) { throw e; }
        if (retryable && attempt === 0) {
          if (["set_captain","remove_player"].includes(command.type) &&
              !state.hostRoster?.some(p=>p.id===command.playerId))throw e;
          if (command.type === "advance" &&
            !(state.phase === "advisers" && state.teams.every(t=>t.locked))) throw e;
          if (command.type === "start" &&
            !(state.phase === "lobby" && state.teams.every(t=>t.ready))) throw e;
          if (command.type === "lock_vote") {
            const team = state.teams[state.viewer.teamIndex];
            // The initial vote may already have landed before the revision conflict.
            if (team.locked || state.privateTeam.voting.myVote) return;
            const runoff = state.privateTeam.voting.runoff;
            if (runoff && !runoff.choices.some(x => x.planIndex === command.planIndex &&
              x.signalIndex === command.signalIndex &&
              x.allocation.amount === command.allocation.amount &&
              x.allocation.route === command.allocation.route)) throw e;
          }
          if (command.type === "lock_advisers" || command.type === "submit_decision") {
            const team = state.teams[state.viewer.teamIndex];
            // If the previous lock already landed, never apply it twice.
            if (team.locked) return;
            if (state.privateTeam.advisers.length !== 2) throw e;
          }
          continue;
        }
        throw e;
      }
    }
  }

  function stopAccess(message) {
    dropSession();
    clearTimeout(retryHandle);
    if (socket && [WebSocket.OPEN, WebSocket.CONNECTING].includes(socket.readyState)) socket.close();
    setConnection(false);
    if (isProjector) {
      history.replaceState(null,"",location.pathname+location.search);
      show("join"); $("join-form").hidden = true;
      $("join-caption").textContent = "This projector link is invalid or has been rotated. Ask the host to generate a new one.";
    } else if (location.pathname === "/join") {
      lobby();
    } else {
      show("create");
    }
    error(message || "Room access is no longer valid.");
  }
  function connect() {
    if (!credential || !roomCode || !navigator.onLine) return;
    if (socket && [WebSocket.OPEN, WebSocket.CONNECTING].includes(socket.readyState)) return;
    clearTimeout(retryHandle);
    try {
      const ws = new WebSocket((location.protocol === "https:" ? "wss://" : "ws://") +
        location.host + "/api/rooms/" + roomCode + "/socket");
      socket = ws;
      setConnection(false);
      ws.addEventListener("open", () => {
        if (socket !== ws || !credential) return;
        ws.send(JSON.stringify({type:"authenticate",token:credential}));
      });
      ws.addEventListener("message", event => {
        if (socket !== ws) return;
        let data;
        try { data = JSON.parse(event.data); } catch (_) { return; }
        if (data.type === "snapshot") {
          retryDelay = 1000; setConnection(true);
          showRoom(data.state);
        } else if (data.type === "error") error(data.error);
      });
      ws.addEventListener("close", evt => {
        if (socket !== ws) return;
        socket = null;
        setConnection(false);
        if (evt.code === 1008) {stopAccess("Access revoked or authentication failed.");return;}
        if (!credential) return;
        retryHandle = setTimeout(async () => {
          if (!credential || !navigator.onLine) return;
          try {showRoom((await getState()).state);}
          catch (e) {
            if (e.httpStatus === 401 || e.httpStatus === 410) {stopAccess(e.message);return;}
          }
          connect();
        }, retryDelay);
        retryDelay = Math.min(retryDelay * 2, 15000);
      });
      ws.addEventListener("error", () => setConnection(false));
    } catch (e) {setConnection(false);error("Live connection unavailable: " + e.message);}
  }
  async function refreshConnection() {
    if (!credential || !roomCode || !navigator.onLine) return;
    try {
      showRoom((await getState()).state);
      if (socket?.readyState === WebSocket.OPEN) socket.send(JSON.stringify({type: "snapshot"}));
      else connect();
    } catch (e) {
      if (e.httpStatus === 401 || e.httpStatus === 410) stopAccess(e.message);
      else setConnection(false);
    }
  }
  async function authenticateSession() {
    try {
      const restored=(await getState()).state;
      // Some browsers duplicate sessionStorage when a room tab is copied.
      // /join is always an invitation to play, never a shortcut back into
      // the host console. If a host credential was inherited, discard the
      // clone only in this tab and show the normal join form.
      if(location.pathname==="/join"&&restored.viewer.role==="host"){
        dropSession();await lobby();return;
      }
      showRoom(restored);
      connect();
    } catch (e) {
      if (e.httpStatus === 401 || e.httpStatus === 410) { stopAccess(e.message); return; }
      setConnection(false); error("Cannot restore room yet: " + e.message);
    }
  }
